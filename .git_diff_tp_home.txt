+  Future<Map<String, dynamic>> _validateMemberQrScanFallback({
+    required String scannedQr,
+    required Map<String, dynamic> decodedPayload,
+  }) async {
+    final userId = decodedPayload['user_id'] as String?;
+    if (userId == null || userId.isEmpty) {
+      return {'valid': false, 'reason': 'Invalid QR code: missing member ID.'};
+    }
+
+    final exp = decodedPayload['exp'];
+    if (exp is int) {
+      final nowEpoch = DateTime.now().toUtc().millisecondsSinceEpoch ~/ 1000;
+      if (nowEpoch > exp) {
+        return {
+          'valid': false,
+          'reason': 'QR code expired. Ask member to refresh and try again.',
+        };
+      }
+    }
+
+    final row = await SupabaseService.instance.client
+        .from('user_qr_codes')
+        .select('user_id, name, surname, is_active, expires_at')
+        .eq('user_id', userId)
+        .eq('qr_code', scannedQr)
+        .eq('is_active', true)
+        .maybeSingle();
+
+    if (row == null) {
+      return {
+        'valid': false,
+        'reason': 'QR not recognized. Ask member to open a fresh QR code.',
+      };
+    }
+
+    final expiresAtRaw = row['expires_at'] as String?;
+    if (expiresAtRaw != null) {
+      try {
+        final expiresAt = DateTime.parse(expiresAtRaw).toUtc();
+        if (expiresAt.isBefore(DateTime.now().toUtc())) {
+          return {
+            'valid': false,
+            'reason': 'Member QR is inactive or subscription expired.',
+          };
+        }
+      } catch (_) {}
+    }
+
+    return {
+      'valid': true,
+      'member_id': row['user_id'],
+      'name': row['name'] ?? decodedPayload['name'] ?? 'Unknown',
+      'surname': row['surname'] ?? decodedPayload['surname'] ?? 'Unknown',
+      'reason': null,
+    };
+  }
